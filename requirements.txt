# face recognition
dlib-bin==20.0.1 #prebuilt binary of dlib, the C++ library that actually does face detection and landmark/encoding extraction
face_recognition_models==0.3.0 #ships the pretrained model files that dlib/face_recognition need 
# face_recognition_models does `from pkg_resources import resource_filename` at
# import time. pkg_resources ships with setuptools, which is NOT installed by
# default in Python 3.12+ environments, and setuptools 81 removes it entirely 
# so this is a real runtime dependency and the upper bound is load-bearing.
setuptools<81 
numpy==2.5.1  #face encodings are numpy arrays
pillow==12.3.0 #image loading/manipulation (opening files, resizing, format conversion) — face_recognition uses it internally too
opencv-python==5.0.0.93 


# liveness / anti-spoofing
# runs attendance/models/antispoof.onnx, which decides whether a detected face is a live
# person or a photo held up to the camera. onnxruntime rather than pytorch because
# running one small model is all that is needed, and pytorch would be a ~2 GB dependency.
onnxruntime==1.28.0
protobuf==7.35.1 # onnxruntime reads the model file, which is a protobuf
flatbuffers==25.12.19 # onnxruntime uses it for its internal serialised format

# timezone database. Windows has no system tz database at all, so zoneinfo cannot resolve
# an IANA name like "Asia/Karachi" without this -- and pinning it means every platform
# agrees on the rules rather than inheriting whatever the host happens to ship.
tzdata==2026.3
